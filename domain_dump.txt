﻿using FriendShip.Domain.Common;
using FriendShip.Domain.Rooms;
using FriendShip.Domain.Scoring;
using FriendShip.Domain.Users;

namespace FriendShip.Domain.Participants;

public class Participant
{
    public const int NameMin = 2;
    public const int NameMax = 30;
    public const int ShareCodeLength = 10;

    /// <summary>Network latency tolerance added to the time limit.</summary>
    public static readonly TimeSpan Grace = TimeSpan.FromSeconds(2);

    public Guid Id { get; private set; }
    public Guid RoomId { get; private set; }
    public Guid VisitorId { get; private set; }
    public Guid? UserId { get; private set; }
    public string DisplayName { get; private set; } = default!;
    public string AvatarKey { get; private set; } = default!;
    public string ShareCode { get; private set; } = default!;
    public DateTime StartedAt { get; private set; }
    public DateTime? CompletedAt { get; private set; }
    public int TotalQuestions { get; private set; }
    public int CorrectCount { get; private set; }
    public int Score { get; private set; }
    public int KnowledgePercent { get; private set; }

    private readonly List<ParticipantAnswer> _answers = new();
    public IReadOnlyList<ParticipantAnswer> Answers => _answers;

    public bool IsCompleted => CompletedAt.HasValue;
    public int AnsweredCount => _answers.Count(a => a.IsFinal);

    private Participant() { }

    public static Participant Join(Room room, int currentParticipants, Guid visitorId, Guid? userId,
        string displayName, string? avatarKey, string shareCode, DateTime now)
    {
        Guard.Against(visitorId == Guid.Empty, "participant.visitor", "شناسه‌ی بازدیدکننده نامعتبر است.");
        Guard.Against(!ShortCode.IsValid(shareCode, ShareCodeLength), "participant.share", "کد اشتراک نامعتبر است.");
        room.EnsureCanJoin(currentParticipants, userId, now);

        return new Participant
        {
            Id = Guid.NewGuid(),
            RoomId = room.Id,
            VisitorId = visitorId,
            UserId = userId,
            DisplayName = Guard.Text(displayName, NameMin, NameMax, "participant.name",
                $"اسمت باید بین {NameMin} تا {NameMax} حرف باشد."),
            AvatarKey = AvatarCatalog.OrDefault(avatarKey),
            ShareCode = shareCode,
            StartedAt = now,
            TotalQuestions = room.Questions.Count
        };
    }

    /// <summary>
    /// Returns the current question (locking its ServedAt on first display). Questions whose time
    /// ran out while the guest was away are recorded as timed out. Returns null when finished.
    /// </summary>
    public ParticipantAnswer? ServeNext(Room room, DateTime now)
    {
        EnsureRoom(room);
        while (!IsCompleted)
        {
            var pending = _answers.FirstOrDefault(a => !a.IsFinal);
            if (pending is null)
            {
                var rq = room.QuestionAt(AnsweredCount + 1)
                         ?? throw new DomainException("participant.question", "سؤال بعدی پیدا نشد.");
                Guard.Against(room.Status != RoomStatus.Open, "room.ended", "مهلت این کوییز تموم شده.");
                pending = new ParticipantAnswer(Id, rq.Id, rq.Order, now);
                _answers.Add(pending);
                return pending;
            }

            if (!IsOverdue(room, pending, now)) return pending;

            pending.Finalize(null, isCorrect: false, timedOut: true, points: 0, now);
            CompleteIfDone(now);
        }
        return null;
    }

    /// <summary>Submits an answer. <paramref name="optionId"/> null means the client timer ran out.</summary>
    public ParticipantAnswer SubmitAnswer(Room room, Guid roomQuestionId, Guid? optionId, DateTime now)
    {
        EnsureRoom(room);
        Guard.Against(IsCompleted, "participant.completed", "این کوییز رو قبلاً تموم کردی.");
        Guard.Against(room.Status != RoomStatus.Open, "room.ended", "مهلت این کوییز تموم شده.");

        var pending = _answers.FirstOrDefault(a => !a.IsFinal);
        Guard.Against(pending is null || pending.RoomQuestionId != roomQuestionId,
            "participant.order", "این سؤال الان قابل جواب دادن نیست.");

        var rq = room.GetQuestion(roomQuestionId);
        if (optionId.HasValue)
            Guard.Against(!rq.HasOption(optionId.Value), "participant.option", "گزینه‌ی انتخاب‌شده معتبر نیست.");

        var elapsed = now - pending!.ServedAt;
        var timedOut = optionId is null || IsOverdue(room, pending, now);
        var selected = timedOut ? null : optionId;
        var correct = !timedOut && rq.IsCorrect(selected);
        var points = ScoreCalculator.PointsFor(correct, room.Settings.Mode, elapsed, room.Settings.TimeLimitSeconds);

        pending.Finalize(selected, correct, timedOut, points, now);
        CompleteIfDone(now);
        return pending;
    }

    private static bool IsOverdue(Room room, ParticipantAnswer a, DateTime now) =>
        room.Settings.TimeLimitSeconds is int limit && now - a.ServedAt > TimeSpan.FromSeconds(limit) + Grace;

    private void CompleteIfDone(DateTime now)
    {
        if (AnsweredCount < TotalQuestions) return;
        var final = _answers.Where(a => a.IsFinal).ToList();
        CorrectCount = final.Count(a => a.IsCorrect);
        Score = final.Sum(a => a.Points);
        KnowledgePercent = ScoreCalculator.Percent(CorrectCount, TotalQuestions);
        CompletedAt = now;
    }

    private void EnsureRoom(Room room) =>
        Guard.Against(room.Id != RoomId, "participant.room", "اتاق نامعتبر است.");
}

public class ParticipantAnswer
{
    public Guid Id { get; private set; }
    public Guid ParticipantId { get; private set; }
    public Guid RoomQuestionId { get; private set; }
    public int Order { get; private set; }
    public Guid? SelectedOptionId { get; private set; }
    public bool IsCorrect { get; private set; }
    public bool TimedOut { get; private set; }
    public int Points { get; private set; }
    public DateTime ServedAt { get; private set; }
    public DateTime? AnsweredAt { get; private set; }

    public bool IsFinal => AnsweredAt.HasValue;

    private ParticipantAnswer() { }

    internal ParticipantAnswer(Guid participantId, Guid roomQuestionId, int order, DateTime servedAt)
    {
        Id = Guid.NewGuid();
        ParticipantId = participantId;
        RoomQuestionId = roomQuestionId;
        Order = order;
        ServedAt = servedAt;
    }

    internal void Finalize(Guid? selectedOptionId, bool isCorrect, bool timedOut, int points, DateTime now)
    {
        SelectedOptionId = selectedOptionId;
        IsCorrect = isCorrect;
        TimedOut = timedOut;
        Points = points;
        AnsweredAt = now;
    }
}

﻿namespace FriendShip.Domain.Scoring;

public sealed record RankInput(Guid ParticipantId, int Score, int CorrectCount, TimeSpan Duration);
public sealed record RankEntry(Guid ParticipantId, int Rank);

public static class Ranking
{
    /// <summary>Competition ranking (1,1,3): higher score, then more correct, then faster. Equal on all = tie.</summary>
    public static IReadOnlyList<RankEntry> Compute(IEnumerable<RankInput> inputs)
    {
        var ordered = inputs
            .OrderByDescending(i => i.Score)
            .ThenByDescending(i => i.CorrectCount)
            .ThenBy(i => i.Duration)
            .ToList();

        var result = new List<RankEntry>(ordered.Count);
        for (var i = 0; i < ordered.Count; i++)
        {
            var rank = i + 1;
            if (i > 0)
            {
                var p = ordered[i - 1]; var c = ordered[i];
                if (p.Score == c.Score && p.CorrectCount == c.CorrectCount && p.Duration == c.Duration)
                    rank = result[i - 1].Rank;
            }
            result.Add(new RankEntry(ordered[i].ParticipantId, rank));
        }
        return result;
    }

    /// <summary>Percentile needs enough people to be meaningful; otherwise null.</summary>
    public static int? Percentile(int score, IReadOnlyCollection<int> allScores, int minSample = 5)
    {
        if (allScores.Count < minSample) return null;
        var below = allScores.Count(s => s < score);
        return (int)Math.Round(below * 100.0 / allScores.Count, MidpointRounding.AwayFromZero);
    }
}

===== Scoring\ScoreCalculator.cs 
﻿using FriendShip.Domain.Rooms;

namespace FriendShip.Domain.Scoring;

public enum ScoreTier { Stranger = 0, Acquaintance = 1, Friend = 2, GoodFriend = 3, BestFriend = 4, SoulMate = 5 }

public static class ScoreCalculator
{
    public const int MaxPointsPerQuestion = 100;
    public const int MinCorrectSpeedPoints = 80;

    public static int Percent(int correct, int total) =>
        total <= 0 ? 0 : (int)Math.Round(correct * 100.0 / total, MidpointRounding.AwayFromZero);

    /// <summary>Classic: 100 per correct answer. Speed: 80–100 depending on how fast.</summary>
    public static int PointsFor(bool correct, GameMode mode, TimeSpan elapsed, int? timeLimitSeconds)
    {
        if (!correct) return 0;
        if (mode == GameMode.Classic || timeLimitSeconds is null or <= 0) return MaxPointsPerQuestion;

        var ratio = Math.Clamp(elapsed.TotalSeconds / timeLimitSeconds.Value, 0, 1);
        var bonus = (MaxPointsPerQuestion - MinCorrectSpeedPoints) * (1 - ratio);
        return MinCorrectSpeedPoints + (int)Math.Round(bonus, MidpointRounding.AwayFromZero);
    }

    public static ScoreTier TierFor(int percent) => percent switch
    {
        >= 100 => ScoreTier.SoulMate,
        >= 81 => ScoreTier.BestFriend,
        >= 61 => ScoreTier.GoodFriend,
        >= 41 => ScoreTier.Friend,
        >= 21 => ScoreTier.Acquaintance,
        _ => ScoreTier.Stranger
    };
}

﻿using FriendShip.Domain.Common;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace FriendShip.Domain.Users
{
    public enum UserRole { User = 0, Admin = 1 }

    public class AppUser
    {
        public const int DisplayNameMin = 2;
        public const int DisplayNameMax = 30;
        public const int EmailMax = 256;
        public const int PublicIdLength = 10;
        public const int BioMax = 160;

        public Guid Id { get; private set; }
        public string PublicId { get; private set; } = default!;
        public string Email { get; private set; } = default!;
        public string DisplayName { get; private set; } = default!;
        public string PasswordHash { get; private set; } = default!;
        public UserRole Role { get; private set; }
        public string AvatarKey { get; private set; } = AvatarCatalog.DefaultKey;
        public string? Bio { get; private set; }
        public bool IsProfilePublic { get; private set; }
        public bool IsBanned { get; private set; }
        public string? BanReason { get; private set; }

        /// <summary>Rotated on password change / ban; cookies with an old stamp are rejected.</summary>
        public string SecurityStamp { get; private set; } = default!;
        public DateTime CreatedAt { get; private set; }
        public DateTime? LastLoginAt { get; private set; }

        public bool IsAdmin => Role == UserRole.Admin;

        private AppUser() { }

        public static AppUser Register(string email, string displayName, string passwordHash, string publicId, DateTime now)
        {
            var normalized = NormalizeEmail(email);
            Guard.Against(normalized.Length is < 5 or > EmailMax || !normalized.Contains('@') || normalized.StartsWith('@'),
                "user.email", "ایمیل معتبر نیست.");
            Guard.Against(string.IsNullOrWhiteSpace(passwordHash), "user.password", "رمز عبور الزامی است.");
            Guard.Against(publicId is null || publicId.Length != PublicIdLength, "user.publicId", "شناسه‌ی عمومی نامعتبر است.");

            return new AppUser
            {
                Id = Guid.NewGuid(),
                PublicId = publicId!,
                Email = normalized,
                DisplayName = ValidName(displayName),
                PasswordHash = passwordHash,
                Role = UserRole.User,
                IsProfilePublic = true,
                SecurityStamp = NewStamp(),
                CreatedAt = now
            };
        }

        public static string NormalizeEmail(string? email) => (email ?? string.Empty).Trim().ToLowerInvariant();

        public void UpdateProfile(string displayName, string? avatarKey, string? bio, bool isPublic)
        {
            DisplayName = ValidName(displayName);
            Guard.Against(!AvatarCatalog.IsValid(avatarKey), "user.avatar", "آواتار انتخاب‌شده معتبر نیست.");
            AvatarKey = avatarKey!;
            var cleanBio = Guard.Normalize(bio);
            Guard.Against(cleanBio.Length > BioMax, "user.bio", $"بیو حداکثر {BioMax} حرف می‌تواند باشد.");
            Bio = cleanBio.Length == 0 ? null : cleanBio;
            IsProfilePublic = isPublic;
        }

        public void ChangePassword(string newHash)
        {
            Guard.Against(string.IsNullOrWhiteSpace(newHash), "user.password", "رمز عبور الزامی است.");
            PasswordHash = newHash;
            SecurityStamp = NewStamp();
        }

        /// <summary>Silent rehash (e.g. hasher upgrade) — does not log the user out.</summary>
        public void UpgradePasswordHash(string newHash) => PasswordHash = newHash;

        public void Ban(string? reason)
        {
            Guard.Against(IsAdmin, "user.ban.admin", "مدیر را نمی‌توان مسدود کرد.");
            IsBanned = true;
            BanReason = Guard.Normalize(reason) is { Length: > 0 } r ? r[..Math.Min(r.Length, 200)] : null;
            SecurityStamp = NewStamp();
        }

        public void Unban() { IsBanned = false; BanReason = null; }

        public void PromoteToAdmin() => Role = UserRole.Admin;
        public void DemoteToUser() => Role = UserRole.User;
        public void MarkLogin(DateTime now) => LastLoginAt = now;

        private static string ValidName(string? name) =>
            Guard.Text(name, DisplayNameMin, DisplayNameMax, "user.name",
                $"نام نمایشی باید بین {DisplayNameMin} تا {DisplayNameMax} حرف باشد.");

        private static string NewStamp() => Guid.NewGuid().ToString("N");
    }
}

﻿using FriendShip.Domain.Common;
using FriendShip.Domain.Questions;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace FriendShip.Domain.Rooms
{
    public class Room
    {
        public const int CodeLength = 6;
        public const int TitleMin = 3;
        public const int TitleMax = 60;
        public static readonly TimeSpan MinLifetime = TimeSpan.FromHours(1);
        public static readonly TimeSpan MaxLifetime = TimeSpan.FromDays(30);
        public static readonly TimeSpan DefaultLifetime = TimeSpan.FromDays(7);

        public Guid Id { get; private set; }
        public string Code { get; private set; } = default!;
        public Guid OwnerUserId { get; private set; }
        public string Title { get; private set; } = default!;
        public RoomSettings Settings { get; private set; } = default!;
        public RoomStatus Status { get; private set; }
        public DateTime CreatedAt { get; private set; }
        public DateTime? OpenedAt { get; private set; }
        public DateTime? EndedAt { get; private set; }
        public DateTime ExpiresAt { get; private set; }

        /// <summary>Optimistic concurrency token; regenerated on every state change.</summary>
        public Guid Version { get; private set; }

        private readonly List<RoomQuestion> _questions = new();
        public IReadOnlyList<RoomQuestion> Questions => _questions;

        private readonly List<RoomStatusChange> _history = new();
        public IReadOnlyList<RoomStatusChange> History => _history;

        public bool IsEnded => Status is RoomStatus.Closed or RoomStatus.Expired or RoomStatus.Cancelled;

        /// <summary>Whether guests may see correct answers (after they finished).</summary>
        public bool AreAnswersRevealable =>
            Settings.RevealMode == RevealMode.Immediate || Status is RoomStatus.Closed or RoomStatus.Expired;

        private Room() { }

        /// <param name="questions">Bank questions loaded with Category and Options.</param>
        public static Room Create(string code, Guid ownerUserId, string title, RoomSettings settings,
            IReadOnlyList<Question> questions, DateTime now, TimeSpan? lifetime = null)
        {
            Guard.Against(!ShortCode.IsValid(code, CodeLength), "room.code", "کد اتاق معتبر نیست.");
            Guard.Against(ownerUserId == Guid.Empty, "room.owner", "سازنده‌ی اتاق مشخص نیست.");
            var life = lifetime ?? DefaultLifetime;
            Guard.Against(life < MinLifetime || life > MaxLifetime, "room.lifetime", "مدت اعتبار اتاق معتبر نیست.");
            Guard.Against(questions.Count != settings.QuestionCount, "room.count",
                $"باید دقیقاً {settings.QuestionCount} سؤال انتخاب شود.");
            Guard.Against(questions.Select(q => q.Id).Distinct().Count() != questions.Count,
                "room.duplicate", "یک سؤال نمی‌تواند دو بار در اتاق باشد.");
            Guard.Against(questions.Any(q => !q.IsActive), "room.inactive", "یکی از سؤال‌ها دیگر فعال نیست.");

            var room = new Room
            {
                Id = Guid.NewGuid(),
                Code = code,
                OwnerUserId = ownerUserId,
                Title = Guard.Text(title, TitleMin, TitleMax, "room.title", $"عنوان باید بین {TitleMin} تا {TitleMax} حرف باشد."),
                Settings = settings,
                Status = RoomStatus.Draft,
                CreatedAt = now,
                ExpiresAt = now + life,
                Version = Guid.NewGuid()
            };
            for (var i = 0; i < questions.Count; i++)
                room._questions.Add(new RoomQuestion(room.Id, i + 1, questions[i]));
            return room;
        }

        public RoomQuestion GetQuestion(Guid roomQuestionId) =>
            _questions.FirstOrDefault(q => q.Id == roomQuestionId)
            ?? throw new DomainException("room.question", "سؤال در این اتاق پیدا نشد.");

        public RoomQuestion? QuestionAt(int order) => _questions.FirstOrDefault(q => q.Order == order);

        // ---------- Draft phase ----------

        public void SetOwnerAnswer(Guid roomQuestionId, Guid optionId)
        {
            EnsureDraft();
            GetQuestion(roomQuestionId).SetOwnerAnswer(optionId);
            Touch();
        }

        public void ReplaceQuestion(Guid roomQuestionId, Question replacement)
        {
            EnsureDraft();
            Guard.Against(!replacement.IsActive, "room.inactive", "این سؤال فعال نیست.");
            var target = GetQuestion(roomQuestionId);
            Guard.Against(_questions.Any(q => q.Id != roomQuestionId && q.SourceQuestionId == replacement.Id),
                "room.duplicate", "این سؤال قبلاً در اتاق هست.");
            target.ReplaceSource(replacement);
            Touch();
        }

        public void Rename(string title)
        {
            Guard.Against(IsEnded, "room.ended", "اتاق تموم شده و قابل ویرایش نیست.");
            Title = Guard.Text(title, TitleMin, TitleMax, "room.title", $"عنوان باید بین {TitleMin} تا {TitleMax} حرف باشد.");
            Touch();
        }

        public void SetLeaderboardVisibility(bool visible)
        {
            Settings.SetLeaderboard(visible);
            Touch();
        }

        // ---------- State machine ----------

        public void Publish(DateTime now)
        {
            ApplyTime(now);
            EnsureDraft();
            Guard.Against(_questions.Any(q => !q.IsAnsweredByOwner), "room.unanswered", "اول باید به همه‌ی سؤال‌ها جواب بدی.");
            ChangeStatus(RoomStatus.Open, now, null);
            OpenedAt = now;
        }

        public void Close(DateTime now)
        {
            ApplyTime(now);
            Guard.Against(Status != RoomStatus.Open, "room.notOpen", "فقط اتاق باز را می‌شود بست.");
            ChangeStatus(RoomStatus.Closed, now, "بسته‌شده توسط سازنده");
            EndedAt = now;
        }

        public void Cancel(DateTime now, string? reason = null)
        {
            Guard.Against(IsEnded, "room.ended", "این اتاق قبلاً تموم شده.");
            ChangeStatus(RoomStatus.Cancelled, now, reason ?? "لغوشده");
            EndedAt = now;
        }

        /// <summary>Lazy expiration. Returns true if the status changed.</summary>
        public bool ApplyTime(DateTime now)
        {
            if (IsEnded || now < ExpiresAt) return false;
            ChangeStatus(RoomStatus.Expired, now, "پایان زمان");
            EndedAt = ExpiresAt;
            return true;
        }

        /// <summary>Validates that a new friend can join. Call <see cref="ApplyTime"/> first.</summary>
        public void EnsureCanJoin(int currentParticipants, Guid? userId, DateTime now)
        {
            Guard.Against(Status == RoomStatus.Draft, "room.draft", "این کوییز هنوز آماده نشده.");
            Guard.Against(Status == RoomStatus.Cancelled, "room.cancelled", "این کوییز لغو شده.");
            Guard.Against(Status is RoomStatus.Closed or RoomStatus.Expired || now >= ExpiresAt,
                "room.ended", "مهلت این کوییز تموم شده.");
            Guard.Against(userId.HasValue && userId.Value == OwnerUserId, "room.self",
                "نمی‌تونی کوییز خودت رو جواب بدی! لینک رو برای دوستات بفرست.");
            Guard.Against(currentParticipants >= Settings.MaxParticipants, "room.full", "ظرفیت این کوییز پر شده.");
        }

        private void EnsureDraft() =>
            Guard.Against(Status != RoomStatus.Draft, "room.notDraft", "بعد از انتشار، سؤال‌ها و جواب‌ها قابل تغییر نیستند.");

        private void ChangeStatus(RoomStatus to, DateTime now, string? reason)
        {
            _history.Add(new RoomStatusChange(Id, Status, to, now, reason));
            Status = to;
            Touch();
        }

        private void Touch() => Version = Guid.NewGuid();
    }

﻿using FriendShip.Domain.Common;
using FriendShip.Domain.Questions;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace FriendShip.Domain.Rooms
{
    public class RoomQuestion
    {
        public Guid Id { get; private set; }
        public Guid RoomId { get; private set; }
        public int SourceQuestionId { get; private set; }
        public int Order { get; private set; }
        public string Text { get; private set; } = default!;
        public string CategoryName { get; private set; } = default!;
        public string CategoryIcon { get; private set; } = default!;

        /// <summary>The owner's answer. Never sent to a guest until reveal is allowed.</summary>
        public Guid? OwnerAnswerOptionId { get; private set; }

        private readonly List<RoomQuestionOption> _options = new();
        public IReadOnlyList<RoomQuestionOption> Options => _options;

        public bool IsAnsweredByOwner => OwnerAnswerOptionId.HasValue;

        private RoomQuestion() { }

        internal RoomQuestion(Guid roomId, int order, Question source)
        {
            Id = Guid.NewGuid();
            RoomId = roomId;
            Order = order;
            CopyFrom(source);
        }

        internal void ReplaceSource(Question source)
        {
            _options.Clear();
            OwnerAnswerOptionId = null;
            CopyFrom(source);
        }

        private void CopyFrom(Question source)
        {
            SourceQuestionId = source.Id;
            Text = source.Text;
            CategoryName = source.Category?.Name ?? string.Empty;
            CategoryIcon = source.Category?.Icon ?? "❓";
            foreach (var o in source.Options.OrderBy(o => o.Order))
                _options.Add(new RoomQuestionOption(Id, o.Order, o.Text));
        }

        public bool HasOption(Guid optionId) => _options.Any(o => o.Id == optionId);

        public bool IsCorrect(Guid? optionId) => optionId.HasValue && optionId == OwnerAnswerOptionId;

        internal void SetOwnerAnswer(Guid optionId)
        {
            Guard.Against(!HasOption(optionId), "room.answer", "گزینه‌ی انتخاب‌شده معتبر نیست.");
            OwnerAnswerOptionId = optionId;
        }
    }

    public class RoomQuestionOption
    {
        public Guid Id { get; private set; }
        public Guid RoomQuestionId { get; private set; }
        public int Order { get; private set; }
        public string Text { get; private set; } = default!;

        private RoomQuestionOption() { }

        internal RoomQuestionOption(Guid roomQuestionId, int order, string text)
        {
            Id = Guid.NewGuid();
            RoomQuestionId = roomQuestionId;
            Order = order;
            Text = text;
        }
    }

    public class RoomStatusChange
    {
        public long Id { get; private set; }
        public Guid RoomId { get; private set; }
        public RoomStatus From { get; private set; }
        public RoomStatus To { get; private set; }
        public DateTime At { get; private set; }
        public string? Reason { get; private set; }

        private RoomStatusChange() { }

        internal RoomStatusChange(Guid roomId, RoomStatus from, RoomStatus to, DateTime at, string? reason)
        {
            RoomId = roomId; From = from; To = to; At = at; Reason = reason;
        }
    }
